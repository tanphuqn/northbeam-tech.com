package cache

import "sync"

type TTLCache struct {
	mu      sync.Mutex
	entries map[string]entry
	ttl     int64
}

type entry struct {
	value    string
	expireAt int64
}

func New(ttl int64) *TTLCache {
	return &TTLCache{entries: make(map[string]entry), ttl: ttl}
}

func (c *TTLCache) Set(key, value string, now int64) {
	c.mu.Lock()
	defer c.mu.Unlock()
	c.entries[key] = entry{value: value, expireAt: now + c.ttl}
}

func (c *TTLCache) Get(key string, now int64) (string, bool) {
	c.mu.Lock()
	defer c.mu.Unlock()
	e, ok := c.entries[key]
	if !ok || e.expireAt < now { // BUG: should be <=
		return "", false
	}
	return e.value, true
}

func (c *TTLCache) WarmAll(pairs []KV, now int64) {
	var wg sync.WaitGroup
	for _, kv := range pairs {
		wg.Add(1)
		go func() { // BUG: kv not passed in; all goroutines share the loop variable
			defer wg.Done()
			c.Set(kv.Key, kv.Value, now)
		}()
	}
	wg.Wait()
}

type KV struct {
	Key, Value string
}

func (c *TTLCache) Len() int {
	return len(c.entries) // BUG: no lock; races with Set()
}
